# Runtime dependencies for the `fireants` package itself.
#
# fireants pins `simpleitk==2.2.1`, which has been un-published from PyPI for
# python >=3.12 (only 2.1.0 and 2.3.0+ still have wheels).  We install a
# working SimpleITK here and then install fireants with --no-deps so pip never
# tries to satisfy that pin.  Any SimpleITK 2.x works at runtime.
#
# NOTE: the `typing` PyPI package is deliberately absent.  It is an obsolete
# backport of the stdlib module and installing it on python 3.12 can shadow
# the real `typing`.  fireants lists it as a dependency; --no-deps saves us.

simpleitk>=2.3,<3
hydra-core
matplotlib
nibabel>=4.0
numpy>=1.21,<2
pandas>=1.3
pytest
scikit-image
scipy>=1.7
tqdm
